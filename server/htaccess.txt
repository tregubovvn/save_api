# ==========================================================================
#  Положите этот файл в папку Task_Arhive на сервере под именем «.htaccess»
#  (точка в начале имени обязательна). Работает на хостингах с Apache.
# ==========================================================================

# Запрет просмотра списка файлов в папке (открываются только прямые ссылки)
Options -Indexes

# Отдавать .xml с корректным Content-Type
AddType application/xml .xml

# Файлы в архиве — только данные: PHP в этой папке не выполняется и не нужен
<FilesMatch "\.(php|phtml|phar)$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
    <IfModule !mod_authz_core.c>
        Order deny,allow
        Deny from all
    </IfModule>
</FilesMatch>
